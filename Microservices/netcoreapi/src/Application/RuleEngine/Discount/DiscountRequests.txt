GiveDiscount30NestedOrExample - 
This rule uses an OrElse operator, so it succeeds if either of its nested conditions is true. For example, you can satisfy both: 
• "IsLoyalAndHasGoodSpend": input1.loyaltyFactor > 3 and totalPurchasesToDate is between 50000 and 100000
• "OrHasHighNumberOfTotalOrders": input2.totalOrders > 15

{
  "BasicInfo": {
    "name": "hello",
    "email": "abcy@xyz.com",
    "creditHistory": "good",
    "country": "india",
    "loyaltyFactor": 5,
    "totalPurchasesToDate": 80000
  },
  "OrderInfo": {
    "totalOrders": 16,
    "recurringItems": 2
  },
  "TelemetryInfo": {
    "noOfVisitsPerMonth": 10,
    "percentageOfBuyingToVisit": 15
  }
}

GiveDiscount35NestedAndExample - 
This rule uses an AndAlso operator, so all nested conditions must be true:
• "IsLoyal": input1.loyaltyFactor > 3
• "AndHasTotalPurchased100000": input1.totalPurchasesToDate is at least 100000
• "AndOtherConditions": input2.totalOrders is greater than 15 and input3.noOfVisitsPerMonth is greater than 25

{
  "BasicInfo": {
    "name": "hello",
    "email": "abcy@xyz.com",
    "creditHistory": "good",
    "country": "india",
    "loyaltyFactor": 4,
    "totalPurchasesToDate": 110000
  },
  "OrderInfo": {
    "totalOrders": 16,
    "recurringItems": 2
  },
  "TelemetryInfo": {
    "noOfVisitsPerMonth": 30,
    "percentageOfBuyingToVisit": 15
  }
}
